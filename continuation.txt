     7    * Watches an element entering the viewport via IntersectionObserver and runs
     8    * an animejs stagger-reveal on its children matching the given selector.
     9    *
    10    * Uses only transform + opacity (compositor-only, no layout thrash).
    11    * Respects prefers-reduced-motion by skipping animation and snapping to end state.
    12    *
    13    * Returns the ref to attach to the container element.
    14    */
    15   export function useAnimeEntrance<T extends HTMLElement>(
    16     childSelector = '[data-anime]',
    17     options?: {
    18       translateY?: number;
    19       duration?: number;
    20       stagger?: number;
    21       delay?: number;
    22       easing?: string;
    23     },
    24   ) {
    25     const ref = useRef<T>(null);
    26     const played = useRef(false);
    27
    28     const {
    29       translateY = 28,
    30       duration = 700,
    31       stagger = 60,
    32       delay = 0,
    33       easing = 'cubicBezier(0.22, 1, 0.36, 1)',
    34     } = options ?? {};
    35
    36     useEffect(() => {
    37       const el = ref.current;
    38       if (!el) return;
    39
    40       const prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    41       const targets = Array.from(el.querySelectorAll<HTMLElement>(childSelector));
    42       if (!targets.length) return;
    43
    44       // Set initial hidden state
    45       if (!prefersReduced) {
    46         targets.forEach((t) => {
    47           t.style.opacity = '0';
    48           t.style.transform = `translateY(${translateY}px)`;
    49         });
    50       }
    51
    52       const observer = new IntersectionObserver(
    53         (entries) => {
    54           if (played.current) return;
    55           const entry = entries[0];
    56           if (!entry.isIntersecting) return;
    57           played.current = true;
    58           observer.disconnect();
    59
    60           if (prefersReduced) {
    61             targets.forEach((t) => {
    62               t.style.opacity = '1';
    63               t.style.transform = 'none';
    64             });
    65             return;
    66           }
    67
    68           anime({
    69             targets,
    70             opacity: [0, 1],
    71             translateY: [translateY, 0],
    72             duration,
    73             delay: anime.stagger(stagger, { start: delay }),
    74             easing,
    75           });
    76         },
    77         { threshold: 0.12, rootMargin: '-40px 0px' },
    78       );
    79
    80       observer.observe(el);
    81       return () => observer.disconnect();
    82     }, [childSelector, translateY, duration, stagger, delay, easing]);
    83
    84     return ref;
    85   }
